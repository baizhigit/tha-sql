Raw SQL (read connection, quoted camelCase columns), returns rows {institutionGroupId, institutionId, concession, day, cnt}:
SELECT g."institutionGroupId", c."institutionId",
       COALESCE(con."name", '') AS concession,
       DATE(c."createdAt") AS day, COUNT(*) AS cnt
FROM customers c
JOIN institution_institution_group g ON g."institutionId" = c."institutionId"
LEFT JOIN LATERAL (
    SELECT cc."concessionId" FROM customer_concessions cc
    WHERE cc."customerId" = c."id" AND cc."isActive" = true AND cc."concessionId" NOT IN (0,90)
    ORDER BY cc."id" DESC LIMIT 1
) acc ON true
LEFT JOIN concessions con ON con."id" = acc."concessionId"
